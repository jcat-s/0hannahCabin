import React, { useMemo } from 'react';
import { AreaChart, Area, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';
import { TrendingUp, Calendar, AlertCircle, Sparkles } from "lucide-react";
import { format, startOfMonth, endOfMonth, eachMonthOfInterval, subMonths, isSameMonth, parseISO } from 'date-fns';

interface PredictiveProps {
    bookings: any[];
}

export function Predictive({ bookings }: PredictiveProps) {
    const predictions = useMemo(() => {
        // 1. PREPARE DATA FOR THE LAST 6 MONTHS
        const now = new Date();
        const last6Months = eachMonthOfInterval({
            start: subMonths(now, 5),
            end: now
        });

        const historicalData = last6Months.map(month => {
            const monthBookings = bookings.filter(b => isSameMonth(parseISO(b.checkInDate || b.checkIn), month));
            return {
                month: format(month, 'MMM'),
                revenue: monthBookings.reduce((sum, b) => sum + (Number(b.totalPrice) || 0), 0),
                bookings: monthBookings.length
            };
        });

        // 2. SIMPLE LINEAR FORECAST (Hula para sa susunod na buwan)
        // Kinukuha ang average growth rate ng revenue
        const revenues = historicalData.map(d => d.revenue);
        let totalGrowth = 0;
        for (let i = 1; i < revenues.length; i++) {
            if (revenues[i-1] !== 0) {
                totalGrowth += (revenues[i] - revenues[i-1]) / revenues[i-1];
            }
        }
        const avgGrowth = revenues.length > 1 ? totalGrowth / (revenues.length - 1) : 0;
        const nextMonthForecast = revenues[revenues.length - 1] * (1 + avgGrowth);

        // 3. IDENTIFY PEAK SEASON
        const peakMonth = [...historicalData].sort((a, b) => b.bookings - a.bookings)[0];

        return {
            historicalData,
            nextMonthForecast: nextMonthForecast > 0 ? nextMonthForecast : revenues[revenues.length - 1],
            avgGrowth: avgGrowth * 100,
            peakMonth
        };
    }, [bookings]);

    return (
        <div className="space-y-8 animate-in fade-in slide-in-from-bottom-4 duration-1000">
            {/* HEADER */}
            <div className="flex items-center gap-3 border-l-4 border-zinc-900 pl-4 mb-6">
                <TrendingUp className="text-zinc-900" size={24} />
                <h2 className="text-2xl font-black uppercase tracking-tighter text-zinc-900">
                    Predictive <span className="text-zinc-400 font-light">Analytics</span>
                </h2>
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
                
                {/* 1. REVENUE TREND CHART */}
                <div className="lg:col-span-2 bg-white p-8 rounded-[2rem] border border-zinc-100 shadow-sm min-h-[400px] flex flex-col">
                    <div className="flex justify-between items-start mb-8">
                        <div>
                            <h3 className="text-[11px] font-black uppercase tracking-[0.2em] text-zinc-400 mb-1">Revenue Momentum</h3>
                            <p className="text-sm text-zinc-500 font-medium">Monthly performance tracking & trend projection</p>
                        </div>
                        <div className="bg-emerald-50 px-3 py-1 rounded-full flex items-center gap-1">
                            <TrendingUp size={12} className="text-emerald-600" />
                            <span className="text-[10px] font-black text-emerald-600">+{predictions.avgGrowth.toFixed(1)}% GROWTH</span>
                        </div>
                    </div>

                    <div className="flex-1 w-full">
                        <ResponsiveContainer width="100%" height="100%">
                            <AreaChart data={predictions.historicalData}>
                                <defs>
                                    <linearGradient id="colorRev" x1="0" y1="0" x2="0" y2="1">
                                        <stop offset="5%" stopColor="#D4AF37" stopOpacity={0.3}/>
                                        <stop offset="95%" stopColor="#D4AF37" stopOpacity={0}/>
                                    </linearGradient>
                                </defs>
                                <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#f0f0f0" />
                                <XAxis 
                                    dataKey="month" 
                                    axisLine={false} 
                                    tickLine={false} 
                                    tick={{fontSize: 10, fontWeight: 700, fill: '#a1a1aa'}}
                                    dy={10}
                                />
                                <YAxis hide />
                                <Tooltip 
                                    contentStyle={{ borderRadius: '15px', border: 'none', boxShadow: '0 10px 15px rgba(0,0,0,0.05)', fontSize: '11px', fontWeight: 'bold' }}
                                    formatter={(val: number) => [`₱${val.toLocaleString()}`, 'Revenue']}
                                />
                                <Area 
                                    type="monotone" 
                                    dataKey="revenue" 
                                    stroke="#D4AF37" 
                                    strokeWidth={4}
                                    fillOpacity={1} 
                                    fill="url(#colorRev)" 
                                />
                            </AreaChart>
                        </ResponsiveContainer>
                    </div>
                </div>

                {/* 2. FORECAST CARDS */}
                <div className="space-y-6">
                    {/* NEXT MONTH ESTIMATE */}
                    <div className="bg-[#18181b] p-8 rounded-[2rem] text-white relative overflow-hidden group">
                        <Sparkles className="absolute top-4 right-4 text-zinc-700 group-hover:text-[#D4AF37] transition-colors" size={20} />
                        <h3 className="text-[10px] font-black uppercase tracking-[0.2em] text-zinc-500 mb-4">Forecasted Income</h3>
                        <p className="text-3xl font-black tracking-tighter mb-2">
                            ₱{predictions.nextMonthForecast.toLocaleString(undefined, { maximumFractionDigits: 0 })}
                        </p>
                        <p className="text-[11px] text-zinc-400 leading-relaxed">
                            Estimated revenue for next month based on current growth momentum and booking patterns.
                        </p>
                    </div>

                    {/* PEAK PERFORMANCE */}
                    <div className="bg-white p-8 rounded-[2rem] border border-zinc-100 shadow-sm">
                        <div className="flex items-center gap-3 mb-4">
                            <div className="w-10 h-10 rounded-2xl bg-amber-50 flex items-center justify-center">
                                <Calendar className="text-[#D4AF37]" size={20} />
                            </div>
                            <div>
                                <h3 className="text-[10px] font-black uppercase tracking-[0.2em] text-zinc-400">Peak Season</h3>
                                <p className="text-sm font-black text-zinc-900">{predictions.peakMonth?.month || 'N/A'}</p>
                            </div>
                        </div>
                        <p className="text-[11px] text-zinc-500 leading-relaxed">
                            Highest occupancy recorded in <span className="font-bold text-zinc-900">{predictions.peakMonth?.month}</span> with 
                            <span className="font-bold text-zinc-900"> {predictions.peakMonth?.bookings} confirmed bookings</span>.
                        </p>
                    </div>

                    {/* AI INSIGHT */}
                    <div className="bg-emerald-50 p-6 rounded-[2rem] border border-emerald-100">
                        <div className="flex gap-3">
                            <AlertCircle className="text-emerald-600 shrink-0" size={18} />
                            <div>
                                <h4 className="text-[11px] font-black uppercase text-emerald-700 mb-1">Market Insight</h4>
                                <p className="text-[11px] text-emerald-800 leading-snug">
                                    {predictions.avgGrowth > 0 
                                        ? "Positive trend detected. Consider opening more slots for weekend 'Evening Chill' stays."
                                        : "Booking pace is steady. Try offering mid-week promos to boost weekday occupancy."}
                                </p>
                            </div>
                        </div>
                    </div>
                </div>

            </div>
        </div>
    );
}
