import React, { useMemo, useState } from 'react';
import {
    BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer,
    PieChart, Pie, Cell, AreaChart, Area, Legend, LineChart, Line
} from 'recharts';
import {
    Users, Baby, Dog, DollarSign, TrendingUp, BrainCircuit,
    Lightbulb, Target, CheckCircle, Calendar, Home, Map,
    Zap, AlertCircle, Star, TrendingDown, Sparkles, Flame, Award, FileText
} from "lucide-react";
import { format, parseISO, getDay, eachDayOfInterval, subDays, differenceInDays, isWithinInterval } from 'date-fns';

const COLORS = ['#D4AF37', '#18181b', '#4ade80', '#fb923c', '#818cf8', '#f472b6', '#a3e635', '#06b6d4'];
const FIXED_PH_HOLIDAYS = ["01-01", "04-09", "05-01", "06-12", "08-31", "11-30", "12-25", "12-30", "02-17", "08-21", "11-01", "11-02", "12-08", "12-24", "12-31"];

const checkIsHighRate = (date: Date, dbHolidays: string[] = []): boolean => {
    const day = getDay(date);
    const isFriSatSun = day === 5 || day === 6 || day === 0;
    const monthDay = format(date, "MM-dd");
    const fullDate = format(date, "yyyy-MM-dd");
    return isFriSatSun || FIXED_PH_HOLIDAYS.includes(monthDay) || dbHolidays.includes(fullDate);
};

const classifyDayType = (date: Date, dbHolidays: string[] = []): 'weekday' | 'weekend' | 'holiday' => {
    const day = getDay(date);
    const isHoliday = checkIsHighRate(date, dbHolidays);
    if (FIXED_PH_HOLIDAYS.includes(format(date, "MM-dd")) || dbHolidays.includes(format(date, "yyyy-MM-dd"))) {
        return 'holiday';
    }
    if (day === 0 || day === 6) return 'weekend';
    if (day === 5) return 'weekend';
    return 'weekday';
};

export function Analytics({ bookings }: { bookings: any[] }) {
    const [dateRange, setDateRange] = useState<'all' | 'month' | 'year' | 'custom'>('all');
    const [startDate, setStartDate] = useState<string>('');
    const [endDate, setEndDate] = useState<string>('');

    const analysis = useMemo(() => {
        // Status Distribution should look at ALL bookings passed to the component, not just Confirmed
        const totalBookingsCount = bookings.length;
        const confirmedList = bookings.filter(b => b.status === 'Confirmed' || b.status === 'Approved');
        const pendingList = bookings.filter(b => b.status === 'Pending');
        const rejectedList = bookings.filter(b => b.status === 'Rejected' || b.status === 'Cancelled');

        const statusData = [
            { name: 'Approved', value: confirmedList.length, color: '#4ade80' },
            { name: 'Pending', value: pendingList.length, color: '#fb923c' },
            { name: 'Rejected', value: rejectedList.length, color: '#f87171' }
        ].filter(item => item.value > 0); // Only show statuses that have records

        // For financial and operational analysis, we filter confirmed/approved bookings
        const confirmed = confirmedList;

        const now = new Date();

        const filteredByRange = confirmed.filter(b => {
            const checkInDate = parseISO(b.checkInDate || b.checkIn);

            if (dateRange === 'custom') {
                if (!startDate || !endDate) return true;
                try {
                    const start = parseISO(startDate);
                    const end = parseISO(endDate);
                    return isWithinInterval(checkInDate, { start, end });
                } catch (e) {
                    return true;
                }
            }
            if (dateRange === 'all') return true;
            if (dateRange === 'month') {
                return checkInDate.getMonth() === now.getMonth() && checkInDate.getFullYear() === now.getFullYear();
            }
            if (dateRange === 'year') {
                return checkInDate.getFullYear() === now.getFullYear();
            }
            return true;
        });

        // --- 1. DESCRIPTIVE DATA ---
        const totalRevenue = filteredByRange.reduce((sum, b) => sum + (Number(b.totalPrice) || 0), 0);
        const totalAdults = filteredByRange.reduce((sum, b) => sum + (Number(b.guests) || 0), 0);
        const totalKids = filteredByRange.reduce((sum, b) => sum + (Number(b.kids) || 0), 0);
        const totalPets = filteredByRange.reduce((sum, b) => sum + (Number(b.pets) || 0), 0);

        // Revenue by Cabin with breakdown
        const cabinRevenue = [
            {
                name: 'Ohannah',
                value: filteredByRange.filter(b => b.cabin?.toLowerCase().includes('ohannah')).reduce((s, b) => s + Number(b.totalPrice), 0),
                count: filteredByRange.filter(b => b.cabin?.toLowerCase().includes('ohannah')).length
            },
            {
                name: 'The Dream',
                value: filteredByRange.filter(b => b.cabin?.toLowerCase().includes('dream')).reduce((s, b) => s + Number(b.totalPrice), 0),
                count: filteredByRange.filter(b => b.cabin?.toLowerCase().includes('dream')).length
            }
        ];

        // Stay Type Distribution with revenue
        const stayTypeData = Object.entries(
            filteredByRange.reduce((acc: any, b) => {
                const stayType = b.stayType || 'full';
                if (!acc[stayType]) {
                    acc[stayType] = { count: 0, revenue: 0 };
                }
                acc[stayType].count += 1;
                acc[stayType].revenue += Number(b.totalPrice);
                return acc;
            }, {})
        ).map(([name, data]: any) => ({
            name: name.toUpperCase(),
            count: data.count,
            revenue: data.revenue
        }));

        // Day Type Analysis (Weekday vs Weekend vs Holiday) - Detailed breakdown
        const weekdayBookings = filteredByRange.filter(b => {
            const checkInDate = parseISO(b.checkInDate || b.checkIn);
            const dayType = classifyDayType(checkInDate);
            return dayType === 'weekday';
        });
        const weekendBookings = filteredByRange.filter(b => {
            const checkInDate = parseISO(b.checkInDate || b.checkIn);
            const dayType = classifyDayType(checkInDate);
            return dayType === 'weekend';
        });
        const holidayBookings = filteredByRange.filter(b => {
            const checkInDate = parseISO(b.checkInDate || b.checkIn);
            const dayType = classifyDayType(checkInDate);
            return dayType === 'holiday';
        });

        const dayTypeData = [
            {
                name: 'Weekday',
                value: weekdayBookings.length,
                revenue: weekdayBookings.reduce((s, b) => s + Number(b.totalPrice), 0),
                color: '#4ade80'
            },
            {
                name: 'Weekend',
                value: weekendBookings.length,
                revenue: weekendBookings.reduce((s, b) => s + Number(b.totalPrice), 0),
                color: '#fb923c'
            },
            {
                name: 'Holiday',
                value: holidayBookings.length,
                revenue: holidayBookings.reduce((s, b) => s + Number(b.totalPrice), 0),
                color: '#f472b6'
            }
        ];

        // Average stay length
        const stayLengths = filteredByRange.map(b => {
            const checkIn = parseISO(b.checkInDate || b.checkIn);
            const checkOut = parseISO(b.checkOutDate || b.checkOut);
            return differenceInDays(checkOut, checkIn);
        });
        const avgStayLength = stayLengths.length > 0 ? (stayLengths.reduce((a, b) => a + b, 0) / stayLengths.length).toFixed(1) : 0;
        const maxStayLength = stayLengths.length > 0 ? Math.max(...stayLengths) : 0;
        const minStayLength = stayLengths.length > 0 ? Math.min(...stayLengths) : 0;

        // Revenue breakdown by day type
        const weekdayRevenue = weekdayBookings.reduce((s, b) => s + Number(b.totalPrice), 0);
        const weekendRevenue = weekendBookings.reduce((s, b) => s + Number(b.totalPrice), 0);
        const holidayRevenue = holidayBookings.reduce((s, b) => s + Number(b.totalPrice), 0);

        // Guest composition
        const avgGuestsPerBooking = filteredByRange.length > 0 ? (totalAdults / filteredByRange.length).toFixed(1) : 0;
        const petFriendlyCount = filteredByRange.filter(b => b.pets > 0).length;
        const familyBookings = filteredByRange.filter(b => b.kids > 0).length;

        // --- 2. PREDICTIVE DATA ---
        const predictedNextMonthRev = totalRevenue * 1.25;
        const predictedOccupancy = Math.min(95, (filteredByRange.length / 30) * 110);
        const avgBookingValue = filteredByRange.length > 0 ? (totalRevenue / filteredByRange.length).toFixed(0) : 0;

        // Growth rates
        const weekdayPercentage = filteredByRange.length > 0 ? ((weekdayBookings.length / filteredByRange.length) * 100).toFixed(1) : 0;
        const weekendPercentage = filteredByRange.length > 0 ? ((weekendBookings.length / filteredByRange.length) * 100).toFixed(1) : 0;
        const holidayPercentage = filteredByRange.length > 0 ? ((holidayBookings.length / filteredByRange.length) * 100).toFixed(1) : 0;

        // --- 3. PRESCRIPTIVE ADVICE ---
        const strategies = [];

        // Holiday strategies
        if (holidayRevenue > 0) {
            const holidayAvg = (holidayRevenue / holidayBookings.length).toFixed(0);
            strategies.push({
                title: "HOLIDAY PEAK",
                desc: `High holiday demand detected (₱${holidayRevenue.toLocaleString()} from ${holidayBookings.length} bookings). Implement a 15% surcharge for upcoming public holidays.`,
                icon: Flame
            });
        }

        // Pet-friendly strategy
        if (totalPets > totalAdults * 0.3) {
            strategies.push({
                title: "PET HAVEN",
                desc: `${petFriendlyCount} pet-friendly bookings (${((petFriendlyCount / filteredByRange.length) * 100).toFixed(0)}% of guests). Offer 'Pawsome Stay' pet kit (+₱500) to boost pet fees.`,
                icon: Dog
            });
        }

        // Family-focused strategy
        if (familyBookings > filteredByRange.length * 0.3) {
            strategies.push({
                title: "FAMILY FOCUS",
                desc: `Kids present in ${familyBookings} bookings (${((familyBookings / filteredByRange.length) * 100).toFixed(0)}%). Add playground corner, kids' activities, or family meal packages.`,
                icon: Baby
            });
        }

        // Weekday promotion
        if (weekdayPercentage && parseFloat(weekdayPercentage) < parseFloat(weekendPercentage || '0')) {
            strategies.push({
                title: "WEEKDAY PROMO",
                desc: `Weekends are ${((parseFloat(weekendPercentage || '0') - parseFloat(weekdayPercentage)) / parseFloat(weekdayPercentage || '1') * 100).toFixed(0)}% busier. Launch 'Mid-week Retreat' 20% discount on Mon-Thu.`,
                icon: TrendingUp
            });
        }

        // Cabin performance strategy
        const topCabin = cabinRevenue.reduce((max, curr) => curr.value > max.value ? curr : max);
        const otherCabin = cabinRevenue.find(c => c.name !== topCabin.name);
        if (otherCabin && topCabin.value > otherCabin.value * 1.5) {
            strategies.push({
                title: "BOOST LOWER CABIN",
                desc: `${topCabin.name} outperforms ${otherCabin.name} by 50%. Consider targeted marketing or competitive pricing for ${otherCabin.name}.`,
                icon: Award
            });
        }

        // Stay length optimization
        if (avgStayLength && parseFloat(avgStayLength as any) < 2) {
            strategies.push({
                title: "EXTEND STAYS",
                desc: `Average stay is only ${avgStayLength} nights. Offer "Stay Longer" discounts: 10% off 3+ nights, 15% off 5+ nights.`,
                icon: Calendar
            });
        }

        // Guest experience strategy
        const highValueBookings = filteredByRange.filter(b => Number(b.totalPrice) > 15000);
        if (highValueBookings.length > 0) {
            strategies.push({
                title: "VIP EXPERIENCE",
                desc: `${highValueBookings.length} premium bookings (₱${(highValueBookings.reduce((s, b) => s + Number(b.totalPrice), 0) / highValueBookings.length).toFixed(0)} avg). Offer concierge, airport transfers, special amenities.`,
                icon: Sparkles
            });
        }

        // Occupancy strategy
        if (predictedOccupancy < 70) {
            strategies.push({
                title: "FILL VACANT DATES",
                desc: `Occupancy forecast: ${predictedOccupancy.toFixed(0)}%. Launch last-minute deals, flash sales, or referral bonuses to reach 85%+ occupancy.`,
                icon: TrendingDown
            });
        }

        // Monthly booking trend for Revenue Trend chart
        const monthlyData: any = {};
        filteredByRange.forEach(b => {
            const month = format(parseISO(b.checkInDate || b.checkIn), 'MMM');
            if (!monthlyData[month]) monthlyData[month] = { month, revenue: 0, count: 0 };
            monthlyData[month].revenue += Number(b.totalPrice);
            monthlyData[month].count += 1;
        });
        const bookingTrendData = Object.values(monthlyData);

        return {
            totalRevenue, totalAdults, totalKids, totalPets,
            cabinRevenue, stayTypeData, dayTypeData, statusData,
            predictedNextMonthRev, predictedOccupancy, strategies,
            avgStayLength, maxStayLength, minStayLength,
            weekdayRevenue, weekendRevenue, holidayRevenue,
            weekdayPercentage, weekendPercentage, holidayPercentage,
            avgGuestsPerBooking, petFriendlyCount, familyBookings,
            avgBookingValue, weekdayBookings, weekendBookings, holidayBookings,
            bookingTrendData
        };
    }, [bookings, dateRange, startDate, endDate]);

    const handleCustomDateChange = (type: 'start' | 'end', val: string) => {
        setDateRange('custom');
        if (type === 'start') setStartDate(val);
        if (type === 'end') setEndDate(val);
    };

    return (
        <div className="p-6 lg:p-10 space-y-12 bg-[#f8f9fa] min-h-screen font-sans text-zinc-900">

            {/* DATE RANGE FILTER */}
            <div className="flex flex-col md:flex-row md:items-center gap-6 p-6 bg-white rounded-3xl border border-zinc-100 shadow-sm justify-between">
                <div className="flex items-center gap-3 flex-wrap">
                    <span className="text-xs font-black text-zinc-400 uppercase tracking-widest">Quick Filter:</span>
                    <button
                        onClick={() => { setDateRange('all'); setStartDate(''); setEndDate(''); }}
                        className={`px-4 py-2 rounded-full text-xs font-black uppercase tracking-widest transition ${dateRange === 'all' ? 'bg-[#D4AF37] text-white' : 'bg-zinc-50 border border-zinc-100 text-zinc-600 hover:border-[#D4AF37]'}`}
                    >
                        All Time
                    </button>
                    <button
                        onClick={() => { setDateRange('month'); setStartDate(''); setEndDate(''); }}
                        className={`px-4 py-2 rounded-full text-xs font-black uppercase tracking-widest transition ${dateRange === 'month' ? 'bg-[#D4AF37] text-white' : 'bg-zinc-50 border border-zinc-100 text-zinc-600 hover:border-[#D4AF37]'}`}
                    >
                        This Month
                    </button>
                    <button
                        onClick={() => { setDateRange('year'); setStartDate(''); setEndDate(''); }}
                        className={`px-4 py-2 rounded-full text-xs font-black uppercase tracking-widest transition ${dateRange === 'year' ? 'bg-[#D4AF37] text-white' : 'bg-zinc-50 border border-zinc-100 text-zinc-600 hover:border-[#D4AF37]'}`}
                    >
                        This Year
                    </button>
                </div>

                <div className="flex items-center gap-3 flex-wrap">
                    <span className="text-xs font-black text-zinc-400 uppercase tracking-widest">Custom Dates:</span>
                    <div className="flex items-center gap-2">
                        <input
                            type="date"
                            value={startDate}
                            onChange={(e) => handleCustomDateChange('start', e.target.value)}
                            className="bg-zinc-50 text-xs font-bold border border-zinc-200 rounded-lg p-2 focus:outline-none focus:ring-1 focus:ring-[#D4AF37] focus:border-[#D4AF37]"
                        />
                        <span className="text-zinc-400 text-xs font-bold">to</span>
                        <input
                            type="date"
                            value={endDate}
                            onChange={(e) => handleCustomDateChange('end', e.target.value)}
                            className="bg-zinc-50 text-xs font-bold border border-zinc-200 rounded-lg p-2 focus:outline-none focus:ring-1 focus:ring-[#D4AF37] focus:border-[#D4AF37]"
                        />
                    </div>
                </div>
            </div>

            {/* KPI TOP ROW */}
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-6 gap-6">
                <KPICard title="Gross Revenue" value={`₱${analysis.totalRevenue.toLocaleString()}`} icon={<DollarSign />} color="text-emerald-600" bg="bg-emerald-50" />
                <KPICard title="Total Bookings" value={analysis.weekdayBookings.length + analysis.weekendBookings.length + analysis.holidayBookings.length} icon={<Calendar />} color="text-orange-600" bg="bg-orange-50" />
                <KPICard title="Total Adults" value={analysis.totalAdults} icon={<Users />} color="text-blue-600" bg="bg-blue-50" />
                <KPICard title="Total Kids" value={analysis.totalKids} icon={<Baby />} color="text-purple-600" bg="bg-purple-50" />
                <KPICard title="Total Pets" value={analysis.totalPets} icon={<Dog />} color="text-amber-600" bg="bg-amber-50" />
                <KPICard title="Avg Stay" value={`${analysis.avgStayLength}N`} icon={<Home />} color="text-pink-600" bg="bg-pink-50" />
            </div>

            {/* --- DESCRIPTIVE SECTION --- */}
            <section className="space-y-6">
                <div className="flex items-center gap-3 border-l-4 border-[#D4AF37] pl-4">
                    <Map className="text-[#D4AF37]" size={24} />
                    <h2 className="text-2xl font-black uppercase tracking-tighter">Descriptive <span className="text-zinc-400 font-light">Report</span></h2>
                </div>

                <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
                    {/* Stay Type Split */}
                    <ChartContainer title="Stay Category Breakdown">
                        <ResponsiveContainer width="100%" height="100%">
                            <BarChart data={analysis.stayTypeData}>
                                <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#eee" />
                                <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{ fontSize: 10, fontWeight: 'bold' }} />
                                <YAxis hide />
                                <Tooltip
                                    contentStyle={{ borderRadius: '10px', border: 'none', boxShadow: '0 10px 30px rgba(0,0,0,0.1)' }}
                                    formatter={(value, name) => name === 'count' ? `${value} bookings` : `₱${value.toLocaleString()}`}
                                />
                                <Bar dataKey="count" fill="#D4AF37" radius={[10, 10, 0, 0]} />
                            </BarChart>
                        </ResponsiveContainer>
                    </ChartContainer>

                    {/* Day Type Analysis */}
                    <ChartContainer title="Bookings by Day Type">
                        <ResponsiveContainer width="100%" height="100%">
                            <PieChart>
                                <Pie data={analysis.dayTypeData.map(d => ({ ...d, value: d.value }))} innerRadius={60} outerRadius={80} paddingAngle={5} dataKey="value">
                                    {analysis.dayTypeData.map((item, i) => <Cell key={i} fill={item.color} />)}
                                </Pie>
                                <Tooltip formatter={(value) => `${value} bookings`} />
                                <Legend />
                            </PieChart>
                        </ResponsiveContainer>
                    </ChartContainer>

                    {/* Booking Status Distribution */}
                    <ChartContainer title="Booking Status Distribution">
                        <ResponsiveContainer width="100%" height="100%">
                            <PieChart>
                                <Pie
                                    data={analysis.statusData}
                                    innerRadius={60}
                                    outerRadius={80}
                                    paddingAngle={5}
                                    dataKey="value"
                                >
                                    {analysis.statusData.map((item, i) => <Cell key={i} fill={item.color} />)}
                                </Pie>
                                <Tooltip formatter={(value) => `${value} stays`} />
                                <Legend verticalAlign="bottom" height={36} />
                            </PieChart>
                        </ResponsiveContainer>
                    </ChartContainer>
                </div>

                <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
                    {/* Revenue by Cabin */}
                    <ChartContainer title="Revenue per Cabin">
                        <ResponsiveContainer width="100%" height="100%">
                            <BarChart data={analysis.cabinRevenue} layout="vertical">
                                <XAxis type="number" hide />
                                <YAxis dataKey="name" type="category" width={100} tick={{ fontSize: 10, fontWeight: 'bold' }} />
                                <Tooltip formatter={(value) => `₱${value.toLocaleString()}`} />
                                <Bar dataKey="value" fill="#18181b" radius={[0, 10, 10, 0]} />
                            </BarChart>
                        </ResponsiveContainer>
                    </ChartContainer>

                    {/* Day Type Revenue */}
                    <ChartContainer title="Revenue by Day Type">
                        <ResponsiveContainer width="100%" height="100%">
                            <BarChart data={analysis.dayTypeData.map(d => ({ name: d.name, value: d.revenue }))}>
                                <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#eee" />
                                <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{ fontSize: 10, fontWeight: 'bold' }} />
                                <YAxis hide />
                                <Tooltip formatter={(value) => `₱${value.toLocaleString()}`} />
                                <Bar dataKey="value" fill="#4ade80" radius={[10, 10, 0, 0]} />
                            </BarChart>
                        </ResponsiveContainer>
                    </ChartContainer>

                    {/* Stay Stats Summary */}
                    <div className="bg-white p-8 rounded-[2.5rem] shadow-sm border border-zinc-100 space-y-6">
                        <h3 className="text-xs font-black uppercase tracking-[0.2em] text-zinc-400 pb-4 border-b border-zinc-50">Stay Statistics</h3>
                        <div className="space-y-4">
                            <div>
                                <p className="text-[10px] font-bold uppercase text-zinc-400 mb-1">Average Length</p>
                                <p className="text-3xl font-black text-[#D4AF37]">{analysis.avgStayLength} <span className="text-lg text-zinc-500">nights</span></p>
                            </div>
                            <div>
                                <p className="text-[10px] font-bold uppercase text-zinc-400 mb-1">Max Length</p>
                                <p className="text-2xl font-black text-zinc-900">{analysis.maxStayLength} nights</p>
                            </div>
                            <div>
                                <p className="text-[10px] font-bold uppercase text-zinc-400 mb-1">Min Length</p>
                                <p className="text-2xl font-black text-zinc-900">{analysis.minStayLength} nights</p>
                            </div>
                            <div className="pt-4 border-t border-zinc-100">
                                <p className="text-[10px] font-bold uppercase text-zinc-400 mb-1">Breakdown</p>
                                <div className="flex flex-col gap-2 text-sm font-bold">
                                    <div className="flex justify-between">
                                        <span className="text-zinc-600">Weekday:</span>
                                        <span className="text-[#D4AF37]">{analysis.weekdayPercentage}%</span>
                                    </div>
                                    <div className="flex justify-between">
                                        <span className="text-zinc-600">Weekend:</span>
                                        <span className="text-[#D4AF37]">{analysis.weekendPercentage}%</span>
                                    </div>
                                    <div className="flex justify-between">
                                        <span className="text-zinc-600">Holiday:</span>
                                        <span className="text-[#D4AF37]">{analysis.holidayPercentage}%</span>
                                    </div>
                                </div>
                            </div>
                        </div>
                    </div>
                </div>
            </section>

            {/* --- PREDICTIVE SECTION --- */}
            <section className="bg-[#18181b] rounded-[3rem] p-8 lg:p-12 text-white overflow-hidden relative">
                <div className="absolute top-0 right-0 p-12 opacity-10">
                    <BrainCircuit size={200} />
                </div>

                <div className="relative z-10 space-y-8">
                    <div className="flex items-center gap-3">
                        <Zap className="text-[#D4AF37]" />
                        <h2 className="text-2xl font-black uppercase tracking-widest italic">Predictive Forecast</h2>
                    </div>

                    <div className="grid grid-cols-1 lg:grid-cols-3 gap-10">
                        <div className="space-y-4">
                            <p className="text-zinc-500 font-bold uppercase tracking-widest text-xs">Revenue Projection</p>
                            <h3 className="text-5xl font-black text-[#D4AF37]">₱{analysis.predictedNextMonthRev.toLocaleString()}</h3>
                            <p className="text-zinc-400 text-sm leading-relaxed">Based on current booking velocity, expect 25% growth in next month's turnover.</p>
                        </div>

                        <div className="space-y-4">
                            <p className="text-zinc-500 font-bold uppercase tracking-widest text-xs">Occupancy Forecast</p>
                            <h3 className="text-5xl font-black text-[#D4AF37]">{analysis.predictedOccupancy.toFixed(0)}%</h3>
                            <p className="text-zinc-400 text-sm leading-relaxed">Estimated occupancy rate for the next period based on current performance.</p>
                        </div>

                        <div className="space-y-4">
                            <p className="text-zinc-500 font-bold uppercase tracking-widest text-xs">Pet-Friendly Bookings</p>
                            <h3 className="text-5xl font-black text-[#D4AF37]">{analysis.petFriendlyCount}</h3>
                            <p className="text-zinc-400 text-sm leading-relaxed">{((analysis.petFriendlyCount / (analysis.weekdayBookings.length + analysis.weekendBookings.length + analysis.holidayBookings.length)) * 100).toFixed(0)}% of guests bring pets.</p>
                        </div>
                    </div>

                    <div className="bg-white/5 rounded-3xl p-6 border border-white/10">
                        <h4 className="text-xs font-bold uppercase tracking-widest mb-6 text-zinc-400">Booking Revenue Over Time</h4>
                        <div className="h-[200px]">
                            <ResponsiveContainer width="100%" height="100%">
                                <AreaChart data={analysis.bookingTrendData}>
                                    <defs>
                                        <linearGradient id="colorRevenue" x1="0" y1="0" x2="0" y2="1">
                                            <stop offset="5%" stopColor="#D4AF37" stopOpacity={0.8} />
                                            <stop offset="95%" stopColor="#D4AF37" stopOpacity={0} />
                                        </linearGradient>
                                    </defs>
                                    <CartesianGrid strokeDasharray="3 3" stroke="rgba(255,255,255,0.1)" />
                                    <XAxis dataKey="month" stroke="rgba(255,255,255,0.5)" />
                                    <YAxis stroke="rgba(255,255,255,0.5)" />

                                    <Area type="monotone" dataKey="revenue" stroke="#D4AF37" fill="url(#colorRevenue)" strokeWidth={3} />
                                </AreaChart>
                            </ResponsiveContainer>
                        </div>
                    </div>
                </div>
            </section>

            {/* --- PRESCRIPTIVE SECTION --- */}
            <section className="space-y-6">
                <div className="flex items-center gap-3 border-l-4 border-blue-500 pl-4">
                    <Lightbulb className="text-blue-500" size={24} />
                    <h2 className="text-2xl font-black uppercase tracking-tighter">Prescriptive <span className="text-zinc-400 font-light">Strategy ({analysis.strategies.length} Recommendations)</span></h2>
                </div>

                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    {analysis.strategies.map((strategy, i) => {
                        const IconComponent = strategy.icon as React.ElementType || Target;
                        return (
                            <div key={i} className="bg-white p-6 rounded-2xl border border-zinc-100 shadow-sm flex items-start gap-4 hover:border-[#D4AF37] transition-all group">
                                <div className="bg-[#D4AF37]/10 p-3 rounded-xl text-[#D4AF37] group-hover:bg-[#D4AF37] group-hover:text-white transition-colors flex-shrink-0">
                                    <IconComponent size={20} />
                                </div>
                                <div className="min-w-0">
                                    <p className="text-xs font-black text-[#D4AF37] uppercase tracking-widest mb-1">{strategy.title}</p>
                                    <p className="text-sm font-medium text-zinc-700 leading-relaxed">{strategy.desc}</p>
                                </div>
                            </div>
                        );
                    })}
                </div>
            </section>
        </div>
    );
}

// Reusable Components for clean code
function KPICard({ title, value, icon, color, bg }: any) {
    return (
        <div className="bg-white p-6 rounded-[2rem] shadow-sm border border-zinc-100 flex items-center justify-between group hover:shadow-xl transition-all">
            <div>
                <p className="text-[10px] font-black uppercase tracking-[0.2em] text-zinc-400 mb-1">{title}</p>
                <h4 className="text-2xl font-black text-zinc-900 tracking-tighter">{typeof value === 'number' ? value.toLocaleString() : value}</h4>
            </div>
            <div className={`p-4 rounded-2xl ${bg} ${color} group-hover:scale-110 transition-transform flex-shrink-0`}>
                {React.cloneElement(icon as React.ReactElement, { size: 24, strokeWidth: 3 } as any)}
            </div>
        </div>
    );
}

function ChartContainer({ title, children }: { title: string, children: React.ReactNode }) {
    return (

            <h3 className="text-xs font-black uppercase tracking-[0.2em] text-zinc-400 mb-8 border-b border-zinc-50 pb-4">{title}</h3>
            <div className="flex-1 min-h-0">
                {children}
            </div>
        </div>
    );
}
